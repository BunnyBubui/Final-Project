import { useState } from 'react';
import { View, Text, TextInput, TouchableOpacity, StyleSheet } from 'react-native';
import { NavigationContainer } from '@react-navigation/native';
import { createNativeStackNavigator } from '@react-navigation/native-stack';

const Stack = createNativeStackNavigator();

function F1() {
  const [u, setU] = useState('');
  const [p, setP] = useState('');
  const [e, setE] = useState('');
  const [n, setN] = useState('');

  return (
    <View style={s.pg}>
      <View style={s.card}>
        <Text style={s.ttl}>ลงทะเบียนผู้ใช้ / แก้ไขโปรไฟล์</Text>

        <View style={s.srchRow}>
          <TextInput placeholder="ค้นหาด้วยชื่อเต็ม / username / email" style={[s.inp, { flex: 1 }]} />
          <TouchableOpacity style={s.srchBtn}>
            <Text style={s.srchTx}>ค้นหา</Text>
          </TouchableOpacity>
        </View>

        <View style={s.grp}>
          <Text style={s.lbl}>Username</Text>
          <TextInput value={u} onChangeText={setU} placeholder="เช่น alice" style={s.inp} />
        </View>

        <View style={s.grp}>
          <Text style={s.lbl}>Password</Text>
          <TextInput value={p} onChangeText={setP} placeholder="อย่างน้อย 6 ตัวอักษร" secureTextEntry style={s.inp} />
        </View>

        <View style={s.grp}>
          <Text style={s.lbl}>Email</Text>
          <TextInput value={e} onChangeText={setE} placeholder="name@example.com" keyboardType="email-address" style={s.inp} />
        </View>

        <View style={s.grp}>
          <Text style={s.lbl}>Full name</Text>
          <TextInput value={n} onChangeText={setN} placeholder="ชื่อ–นามสกุล" style={s.inp} />
        </View>

        <TouchableOpacity style={s.btn}>
          <Text style={s.btnTx}>สร้างผู้ใช้ใหม่</Text>
        </TouchableOpacity>

        <TouchableOpacity style={{ marginTop: 12 }} onPress={() => { setU(''); setP(''); setE(''); setN(''); }}>
          <Text style={{ textAlign: 'center', color: '#6B7280', fontWeight: '600' }}>ล้างแบบฟอร์ม</Text>
        </TouchableOpacity>
      </View>
    </View>
  );
}

function F2({ navigation }) {
  const [u, setU] = useState('');
  const [p, setP] = useState('');
  const [err, setErr] = useState('');

  const onLogin = () => {
    if (!u.trim() || !p.trim()) {
      setErr('กรุณากรอก username และ password');
      return;
    }
    setErr('');
    navigation.replace('Form1');
  };

  return (
    <View style={s.pg}>
      <View style={s.card}>
        <Text style={s.ttl}>เข้าสู่ระบบ</Text>

        <View style={s.grp}>
          <TextInput value={u} onChangeText={setU} placeholder="Username" autoCapitalize="none" style={s.inp} />
        </View>

        <View style={s.grp}>
          <TextInput value={p} onChangeText={setP} placeholder="Password" secureTextEntry style={s.inp} />
        </View>

        {!!err && <Text style={s.err}>{err}</Text>}

        <TouchableOpacity style={s.btn} onPress={onLogin}>
          <Text style={s.btnTx}>Login</Text>
        </TouchableOpacity>
      </View>
    </View>
  );
}

export default function App() {
  return (
    <NavigationContainer>
      <Stack.Navigator initialRouteName="Form2" screenOptions={{ headerShown: false }}>
        <Stack.Screen name="Form2" component={F2} />
        <Stack.Screen name="Form1" component={F1} />
      </Stack.Navigator>
    </NavigationContainer>
  );
}
